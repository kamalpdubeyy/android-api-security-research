========================================
1. PROJECT DESCRIPTION
========================================
This artefact was developed to support the MSc Cybersecurity project invest.

The system contains:
- An Android client application
- A backend REST API built with Node.js and Express
- A SQLite database
- An insecure API version for vulnerability demonstration
- A secure API version for mitigation and comparison

The testing focused on:
- Broken Authentication
- Broken Object Level Authentication (BOLA / IDOR)
- Sensitive Data Exposure
- JWT token enforcement

4. SECURE API VERSION
File: server_secure.js
Purpose: Implements JWT authentication and object-level authorization
========================================
const express = require("express");
const sqlite3 = require("sqlite3").verbose();
const jwt = require("jsonwebtoken");
const bodyParser = require("body-parser");
const cors = require("cors");

const app = express();
app.use(bodyParser.json());
app.use(cors());

const SECRET = "mscproject";

const db = new sqlite3.Database("./database.db");

db.serialize(() => {
    db.run(`
        CREATE TABLE IF NOT EXISTS users(
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            username TEXT,
            password TEXT
        )
    `);

    db.run(`
        INSERT INTO users(username,password)
        SELECT 'user','password'
        WHERE NOT EXISTS (
            SELECT 1 FROM users WHERE username='user'
        )
    `);

    db.run(`
        INSERT INTO users(username,password)
        SELECT 'kamal','onpt66604711'
        WHERE NOT EXISTS (
            SELECT 1 FROM users WHERE username='kamal'
        )
    `);

    db.run(`
        INSERT INTO users(username,password)
        SELECT 'aastha','aastha@4933'
        WHERE NOT EXISTS (
            SELECT 1 FROM users WHERE username='aastha'
        )
    `);
});

app.post("/login", (req, res) => {
    const { username, password } = req.body;

    db.get(
        "SELECT * FROM users WHERE username=? AND password=?",
        [username, password],
        (err, row) => {
            if (err) return res.status(500).send(err.message);
            if (!row) return res.status(401).json({ message: "Invalid credentials" });

            const token = jwt.sign(
                { id: row.id, username: row.username },
                SECRET,
                { expiresIn: "1h" }
            );

            res.json({ token });
        }
    );
});

// authentication middleware
function verifyToken(req, res, next) {
    const authHeader = req.headers["authorization"];

    if (!authHeader) {
        return res.status(401).json({ message: "Token required" });
    }

    const parts = authHeader.split(" ");
    if (parts.length !== 2 || parts[0] !== "Bearer") {
        return res.status(401).json({ message: "Invalid authorization format" });
    }

    const token = parts[1];

    jwt.verify(token, SECRET, (err, decoded) => {
        if (err) {
            return res.status(403).json({ message: "Invalid or expired token" });
        }

        req.user = decoded;
        next();
    });
}

// secure endpoint
app.get("/user/:id", verifyToken, (req, res) => {
    const requestedId = parseInt(req.params.id, 10);

    // object-level authorization
    if (req.user.id !== requestedId) {
        return res.status(403).json({ message: "Forbidden: cannot access another user's data" });
    }

    db.get(
        "SELECT id, username FROM users WHERE id=?",
        [requestedId],
        (err, row) => {
            if (err) return res.status(500).send(err.message);
            if (!row) return res.status(404).json({ message: "User not found" });

            res.json(row);
        }
    );
});

app.listen(3000, () => {
    console.log("Secure server running on port 3000");
});

package com.msc.cyber

import retrofit2.Call
import retrofit2.http.Body
import retrofit2.http.POST

data class LoginRequest(
    val username: String,
    val password: String
)

data class TokenResponse(
    val token: String
)

interface ApiService {
    @POST("login")
    fun login(@Body request: LoginRequest): Call<TokenResponse>
}

package com.msc.cyber

import retrofit2.Retrofit
import retrofit2.converter.gson.GsonConverterFactory

object RetrofitClient {

    private const val BASE_URL = "http://10.0.2.2:3000/"

    val api: ApiService by lazy {
        Retrofit.Builder()
            .baseUrl(BASE_URL)
            .addConverterFactory(GsonConverterFactory.create())
            .build()
            .create(ApiService::class.java)
    }
}

7. ANDROID MAIN ACTIVITY CODE
File: MainActivity.kt
Purpose: Sends login request and prints JWT token in Logcat

package com.msc.cyber

import android.os.Bundle
import android.util.Log
import androidx.activity.ComponentActivity
import androidx.activity.compose.setContent
import retrofit2.Call
import retrofit2.Callback
import retrofit2.Response

class MainActivity : ComponentActivity() {

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)

        setContent { }

        RetrofitClient.api.login(
            LoginRequest("user", "password")
        ).enqueue(object : Callback<TokenResponse> {

            override fun onResponse(
                call: Call<TokenResponse>,
                response: Response<TokenResponse>
            ) {
                Log.d("API_TEST", "TOKEN: ${response.body()?.token}")
            }

            override fun onFailure(
                call: Call<TokenResponse>,
                t: Throwable
            ) {
                Log.e("API_TEST", "ERROR: ${t.message}")
            }
        })
    }
}

========================================
11. SECURITY FEATURES IMPLEMENTED
========================================

The secure API implements the following controls:

1. JWT-based authentication
2. Token validation middleware
3. Object-level authorization
4. Restricted access to user-specific records
5. Removal of password field from API response
6. Appropriate HTTP status codes:
   - 401 Unauthorized for missing token
   - 403 Forbidden for unauthorized object access
   - 200 OK for valid authorized access
